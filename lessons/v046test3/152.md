讓 Agent **向前移動**。當 Agent **檢查下方**方塊，發現**不是岩漿塊**時，就持續**向下移動**。

player.onChat("magma", function () {
    agent.move(FORWARD, 1)
    while (agent.inspect(AgentInspection.Block, DOWN) != MAGMA_BLOCK) {
        agent.move(DOWN, 1)
    }
})
