## 步驟 1
程式提供三個函式：``||functions: plantSeed||``、``||functions: plantSection||`` 和 ``||functions: checkTurn||``。先建立新的 ``||player: 當收到聊天指令時||`` 指令，再加入條件：``||loops:當條件成立時重複||`` Agent 執行 ``||agent:檢查下方方塊||``，而且不是**黃金方塊**時，就 ``||functions: 呼叫||`` 所需的函式。 



```template
/**
 * We are calling a function inside a function
 */
function plantSection () {
    for (let index = 0; index < 11; index++) {
        plantSeed()
    }
    agent.move(FORWARD, 1)
}
 /**
 * The code was modified to not place seeds if there's no block under the Agent.
 */
function plantSeed () {
    agent.till(FORWARD)
    agent.move(FORWARD, 1)
    if (agent.detect(AgentDetection.Block, DOWN)) {
        agent.place(DOWN)
    }
}
function checkTurn () {
    if (agent.inspect(AgentInspection.Block, DOWN) == LAPIS_LAZULI_BLOCK) {
        agent.turn(RIGHT_TURN)
        agent.move(FORWARD, 1)
        agent.turn(RIGHT_TURN)
    } else if (agent.inspect(AgentInspection.Block, DOWN) == BLOCK_OF_QUARTZ) {
        agent.turn(LEFT_TURN)
        agent.move(FORWARD, 1)
        agent.turn(LEFT_TURN)
    }
}
```
